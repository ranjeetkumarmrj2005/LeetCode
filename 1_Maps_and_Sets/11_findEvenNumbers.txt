class Solution {
public:
    vector<int> findEvenNumbers(vector<int>& digits) {
        vector<int>v;
        unordered_map<int,int>m;
        for(auto d: digits){
            m[d]++;
        }
        for(int i=100;i<=998;i=i+2){

            int a=i%10;
            int b=(i/10)%10;
            int c=i/100;

            m[a]--;
            m[b]--;
            m[c]--;

            if(m[a]>=0 && m[b]>=0 && m[c]>=0){
                v.push_back(i);
            }
            m[a]++;
            m[b]++;
            m[c]++;
        }
        return v;
    }
};
