class Solution {
public:
    vector<int> findEvenNumbers(vector<int>& digits) {
        vector<int>v;
        unordered_map<int,int>m;
        for(auto x:digits){
            m[x]++;
        }
        for(int i=100;i<=999;i=i+2){
            int x=i;
            int a=x%10;
            x=x/10;
            int b=x%10;
            x=x/10;
            int c=x;
            if(m.find(a)!=m.end()){
                m[a]--;
                if(m[a]==0) m.erase(a);
                if(m.find(b)!=m.end()){
                    m[b]--;
                    if(m[b]==0) m.erase(b);
                    if(m.find(c)!=m.end()) v.push_back(i);
                    m[b]++;
                }
                m[a]++;
            }
        }
        return v;
    }
};
