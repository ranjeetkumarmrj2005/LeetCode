class Solution {
public:
    bool canArrange(vector<int>& arr, int k) {
        int n=arr.size();
        unordered_map<int,int>m;
        vector<int>v;
        for(int i=0;i<n;i++){
            int rem=arr[i]%k;
            if(rem<0){
                m[rem+k]++;
            }
            else{
                m[rem]++;
            }
        }
        for(auto x:m){
            int key=x.first;
            int freq=x.second;
            int complement=k-key;
            int complementfreq;
            if(key==0){
                if(freq % 2 !=0) return false;
                continue;
            }
            if(k%2==0 && key==k/2){
                if(freq%2!=0) return false;
                continue;
            }
            if(m.find(complement)!=m.end()) {
                complementfreq=m[complement];
            }
            else return false;
            if (freq!=complementfreq) return false;
        }
        return true;
    }
};
