class Solution {
public:
    int subarraySum(vector<int>& nums, int k) {
        int n=nums.size();
        vector<int>v(n,0);
        v[0]=nums[0];
        for(int i=1;i<n;i++){
            v[i]=v[i-1]+nums[i];
        }
        unordered_map<int,int>m;
        int count=0;
        for(int i=0;i<n;i++){
            if(v[i]==k){
                count++;
            }
            if(m.find(v[i]-k)!=m.end()){
                count=count+m[v[i]-k];
            }
            m[v[i]]++;
        }
        return count;
    }
};
