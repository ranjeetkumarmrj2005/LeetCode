class Solution {
public:
    double power(double x, long long n){
        if(n==0) return 1;
        if(n==1) return x;
        if(x==0) return 0;
        if(x==1) return 1;
        double t=power(x,n/2);
        if(n%2==0) return t*t;
        else return t*t*x;

    }
    double myPow(double x, int n) {
        
        double ans;
        long long p=n;
        if(p>0){
            ans=power(x,p);
        }
        else {
            p=-p;
            ans=power(x,p);
            ans=1/ans; 
        }
        return ans;
    }
};
