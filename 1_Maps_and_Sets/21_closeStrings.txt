class Solution {
public:
    bool closeStrings(string str1, string str2) {
        if(str1.size()!=str2.size()) return false;
        int n=str1.length();
        unordered_map<char,int>m1;
        unordered_map<char,int>m2;
        unordered_map<int,int>m1freq;
        unordered_map<int,int>m2freq;

        for(int i=0;i<n;i++){
            m1[str1[i]]++;
            m2[str2[i]]++;
        }
        for(auto &x:m1){
            if(m2.find(x.first)!=m2.end()){
                m1freq[x.second]++;
            }
            else return false;
            
        }
        for(auto &x:m2){
            m2freq[x.second]++;
        }
        for(auto x: m1freq){
            if(m2freq.find(x.first)==m2freq.end()) return false;
            if(m2freq[x.first]!=m1freq[x.first]) return false;
            
        }
        return true;
    }
};
