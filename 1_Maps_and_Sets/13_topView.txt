class Solution {
  public:
    vector<int> topView(Node *root) {
        
       vector<int>ans;
       map<int,int>m;
       if(root==NULL) return ans;
       queue< pair<Node*, int> >q;
       
       q.push({root,0});
       
       while(q.size()>0){
           
           Node* curr=q.front().first;
           int hd= q.front().second;
           q.pop();
           
           if(m.find(hd)==m.end()){
               m[hd]=curr->data;
           }
           if(curr->left!=NULL){
               q.push({curr->left,hd-1});
           }
           if(curr->right!=NULL){
               q.push({curr->right,hd+1});
           }
       }
       for(auto x:m){
           ans.push_back(x.second);
       }
       return ans;
    }
};
