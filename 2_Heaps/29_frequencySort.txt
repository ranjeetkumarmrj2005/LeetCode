class Solution {
public:
    vector<int> frequencySort(vector<int>& nums) {
        int n = nums.size();
        unordered_map<int, int> m;
        
        for(int i = 0; i < n; i++){
            m[nums[i]]++;
        }
        
        vector< pair<int,int> >v;
        for(auto &x:m){
            v.push_back({x.second, x.first});
        }

        sort(v.begin(), v.end(), [](pair<int, int>&a, pair<int,int> &b){
            if(a.first==b.first){
                return a.second > b.second;
            }
            else{
                return a.first < b.first;
            }
        });

        
        sort(v.begin(), v.end(), [](pair<int, int>& a, pair<int, int>& b){
            if(a.first == b.first) {
                return a.second > b.second;
            }
            return a.first < b.first;
        });
        

        vector<int>ans;
        for(int i=0;i<v.size();i++){
            int freq=v[i].first;
            int val=v[i].second;
            for(int j=0;j<freq;j++){
                ans.push_back(val);
            }
        }
        return ans;
        
        // vector<int> ans;
        // for(int i = 0; i < v.size(); i++){
        //     int freq = v[i].first;
        //     int val = v[i].second;
        //     for(int j = 0; j < freq; j++){
        //         ans.push_back(val);
        //     }
        // }
        // return ans;
    }
};
