class Solution {
public:
    int lastStoneWeight(vector<int>& stones) {
        int n=stones.size();
        if(stones.size()==1) return stones[0];
        priority_queue< int> pq;
        for(int i=0;i<n;i++){
            pq.push(stones[i]);
        }
        while(pq.size()>1){
            int max=pq.top();
            pq.pop();
            int smax=pq.top();
            pq.pop();
            if(max!=smax){
                pq.push(max-smax);
            }
        }
        if(pq.size()==1) return pq.top();
        else return 0;
    }
};
