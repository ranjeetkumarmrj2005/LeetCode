class Solution {
  public:
    void nearlySorted(vector<int>& arr, int k) {
        int n=arr.size();
        priority_queue<int, vector<int>, greater<int>> pq;
        for(int i=0;i<n;i++){
            pq.push(arr[i]);
            if(pq.size()>k){
                int top=pq.top();
                arr[i-k]=top;
                pq.pop();
            }
        }
        int i=n-k;
        while(pq.size()>0){
            int top=pq.top();
            pq.pop();
            arr[i]=top;
            i++;
        }
        return ;
    }
};
