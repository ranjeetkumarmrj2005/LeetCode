class Solution {
public:
    int kthSmallest(vector<vector<int>>& matrix, int k) {
        if(k==1) return matrix[0][0];
        int n=matrix.size();
        int m=matrix[0].size();
        priority_queue< int >pq;
        for(int i=0;i<n;i++){
            for(int j=0;j<n;j++){
                pq.push(matrix[i][j]);
                if(pq.size()>k){
                    pq.pop();
                    // return pq.top();
                }
            }
        }
        return pq.top();
    }
};
