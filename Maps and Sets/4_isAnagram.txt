class Solution {
public:
    bool isAnagram(string s, string t) {
        if(s.length()!=t.length()) return false;
        unordered_map<char,int>map1;
        for(int i=0;i<s.length();i++){
            map1[s[i]]++;
        }
        
        for(int i=0;i<t.length();i++){
            if(map1.find(t[i])!=map1.end()){
                map1[t[i]]--;
                if(map1[t[i]]==0) {
                    map1.erase(t[i]);
                }
            }
            else return false;
        }
        return true;
    }
};
